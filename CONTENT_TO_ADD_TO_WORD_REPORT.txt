Vehicle Detector Implementation:

from ultralytics import YOLO
import cv2
import numpy as np
from typing import List, Dict
import logging

class VehicleDetector:
    """Real-time vehicle detection using YOLOv8"""
    
    def __init__(self, model_name: str = "yolov8m.pt", 
                 confidence_threshold: float = 0.35):
        """
        Initialize the vehicle detector.
        
        Args:
            model_name: YOLOv8 model file (m=medium for parking lots)
            confidence_threshold: Min confidence score (0.35 optimized for parking)
        """
        self.confidence_threshold = confidence_threshold
        self.model = YOLO(model_name)
        
    def detect_vehicles(self, frame: np.ndarray) -> List[Dict]:
        """
        Detect vehicles in a video frame.
        
        Returns:
            List of detections with box coordinates, confidence, and class
        """
        results = self.model(
            frame,
            conf=self.confidence_threshold,
            imgsz=1280,  # Critical for aerial parking views
            iou=0.40,    # NMS threshold for parking lots
            device='cpu'
        )
        
        detections = []
        for result in results:
            for box in result.boxes:
                x1, y1, x2, y2 = box.xyxy[0].cpu().numpy().astype(int)
                conf = float(box.conf[0])
                cls = int(box.cls[0])
                
                class_names = {2: "car", 5: "bus", 7: "truck"}
                class_name = class_names.get(cls, "vehicle")
                
                detections.append({
                    "box": [x1, y1, x2, y2],
                    "confidence": conf,
                    "class_name": class_name,
                    "center": ((x1 + x2) // 2, (y1 + y2) // 2),
                    "area": (x2 - x1) * (y2 - y1)
                })
        
        return detections
    
    def draw_detections(self, frame: np.ndarray, 
                       detections: List[Dict]) -> np.ndarray:
        """Draw bounding boxes on frame"""
        frame_copy = frame.copy()
        for det in detections:
            x1, y1, x2, y2 = det["box"]
            conf = det["confidence"]
            label = f"{det['class_name']} {conf:.2f}"
            
            cv2.rectangle(frame_copy, (x1, y1), (x2, y2), 
                         (0, 255, 0), 2)
            cv2.putText(frame_copy, label, (x1, y1 - 5),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.5, 
                       (0, 255, 0), 2)
        
        return frame_copy


License Plate Recognition Implementation:

import easyocr
import cv2
import numpy as np

class LicensePlateRecognizer:
    """Extract license plate text from vehicle images"""
    
    def __init__(self):
        """Initialize EasyOCR reader"""
        self.reader = easyocr.Reader(['en'], gpu=False)
    
    def extract_plate_region(self, frame: np.ndarray, 
                            box: List[int]) -> np.ndarray:
        """
        Extract license plate region from detected vehicle.
        Plate is typically in lower portion of vehicle bounding box.
        """
        x1, y1, x2, y2 = box
        height = y2 - y1
        
        # License plate region in lower 30% of vehicle
        plate_y1 = y2 - int(height * 0.3)
        plate_region = frame[plate_y1:y2, x1:x2]
        
        return plate_region
    
    def recognize_plate(self, plate_image: np.ndarray) -> str:
        """
        Recognize text from license plate image.
        
        Returns:
            Extracted license plate number
        """
        try:
            # Convert BGR to RGB for EasyOCR
            plate_rgb = cv2.cvtColor(plate_image, cv2.COLOR_BGR2RGB)
            
            # Enhance contrast for better OCR
            lab = cv2.cvtColor(plate_image, cv2.COLOR_BGR2LAB)
            l, a, b = cv2.split(lab)
            clahe = cv2.createCLAHE(clipLimit=3.0, tileGridSize=(8, 8))
            l = clahe.apply(l)
            enhanced = cv2.merge([l, a, b])
            enhanced_bgr = cv2.cvtColor(enhanced, cv2.COLOR_LAB2BGR)
            
            # Run OCR
            results = self.reader.readtext(enhanced_bgr)
            
            # Extract and clean text
            plate_text = "".join([text[1] for text in results])
            plate_text = plate_text.replace(" ", "").upper()
            
            return plate_text if plate_text else "UNKNOWN"
            
        except Exception as e:
            print(f"OCR Error: {e}")
            return "ERROR"


Parking Slot Detection Implementation:

class ParkingSlotDetector:
    """Determine occupancy status of predefined parking slots"""
    
    def __init__(self, slot_coords: List[List[int]]):
        """
        Initialize with predefined parking slot coordinates.
        
        Args:
            slot_coords: List of [x1, y1, x2, y2] coordinates for each slot
        """
        self.slot_coords = slot_coords
        self.slot_status = {}  # Track occupied/vacant status
    
    def detect_occupancy(self, frame: np.ndarray, 
                        detections: List[Dict]) -> Dict[int, str]:
        """
        Determine which slots are occupied based on vehicle detections.
        
        Returns:
            Dictionary mapping slot_id to "occupied" or "vacant"
        """
        slot_status = {}
        
        for idx, (x1, y1, x2, y2) in enumerate(self.slot_coords):
            occupied = False
            
            # Check if any vehicle overlaps with this slot
            for det in detections:
                vx1, vy1, vx2, vy2 = det["box"]
                
                # Calculate intersection area
                ix1 = max(x1, vx1)
                iy1 = max(y1, vy1)
                ix2 = min(x2, vx2)
                iy2 = min(y2, vy2)
                
                if ix2 > ix1 and iy2 > iy1:
                    intersection = (ix2 - ix1) * (iy2 - iy1)
                    slot_area = (x2 - x1) * (y2 - y1)
                    
                    # If >50% overlap, mark as occupied
                    if intersection / slot_area > 0.5:
                        occupied = True
                        break
            
            slot_status[idx] = "Occupied" if occupied else "Vacant"
        
        self.slot_status = slot_status
        return slot_status
    
    def visualize_slots(self, frame: np.ndarray) -> np.ndarray:
        """Draw slot rectangles with color coding"""
        frame_copy = frame.copy()
        
        for idx, (x1, y1, x2, y2) in enumerate(self.slot_coords):
            status = self.slot_status.get(idx, "Vacant")
            color = (0, 0, 255) if status == "Occupied" else (0, 255, 0)
            
            cv2.rectangle(frame_copy, (x1, y1), (x2, y2), color, 2)
            cv2.putText(frame_copy, f"Slot {idx}: {status}", 
                       (x1, y1 - 5),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)
        
        return frame_copy


Complete Pipeline Integration:

class SmartParkingPipeline:
    """End-to-end parking management system"""
    
    def __init__(self, yolo_model: str, slot_coords: List[List[int]]):
        self.detector = VehicleDetector(yolo_model)
        self.ocr = LicensePlateRecognizer()
        self.slot_detector = ParkingSlotDetector(slot_coords)
        self.vehicles_detected = []
    
    def process_frame(self, frame: np.ndarray) -> Dict:
        """
        Process single video frame through complete pipeline.
        
        Returns:
            Dictionary with detections, OCR results, and slot status
        """
        # Step 1: Detect vehicles
        detections = self.detector.detect_vehicles(frame)
        
        # Step 2: Extract license plates
        plate_numbers = []
        for det in detections:
            plate_region = self.ocr.extract_plate_region(
                frame, det["box"])
            plate_text = self.ocr.recognize_plate(plate_region)
            plate_numbers.append(plate_text)
        
        # Step 3: Detect parking slot occupancy
        slot_status = self.slot_detector.detect_occupancy(
            frame, detections)
        
        # Step 4: Calculate statistics
        stats = {
            "total_vehicles": len(detections),
            "occupied_slots": sum(1 for s in slot_status.values() 
                                 if s == "Occupied"),
            "available_slots": len(self.slot_coords) - sum(
                1 for s in slot_status.values() if s == "Occupied"),
            "plate_numbers": plate_numbers,
            "detections": detections,
            "slot_status": slot_status
        }
        
        return stats
    
    def process_video(self, video_path: str) -> Dict:
        """Process entire video and generate report"""
        cap = cv2.VideoCapture(video_path)
        frame_count = 0
        total_vehicles = set()
        parking_log = []
        
        while True:
            ret, frame = cap.read()
            if not ret:
                break
            
            # Process frame
            stats = self.process_frame(frame)
            
            # Accumulate data
            for plate in stats["plate_numbers"]:
                total_vehicles.add(plate)
            
            parking_log.append({
                "frame": frame_count,
                "occupied": stats["occupied_slots"],
                "available": stats["available_slots"]
            })
            
            frame_count += 1
        
        cap.release()
        
        return {
            "total_frames": frame_count,
            "unique_vehicles": len(total_vehicles),
            "parking_log": parking_log
        }

A.1 MODELS.PY - DATABASE SCHEMA

from django.db import models
from django.utils import timezone

class User_details(models.Model):
    User_id = models.AutoField(primary_key=True)
    Email = models.EmailField(max_length=50, unique=True)
    Password = models.CharField(max_length=128, null=True)
    
    class Meta:
        db_table = 'user_details'
    
    def __str__(self):
        return self.Email


class ParkingLot(models.Model):
    """Define parking lots"""
    lot_id = models.AutoField(primary_key=True)
    lot_name = models.CharField(max_length=100, unique=True)
    total_spots = models.IntegerField(default=50)
    created_at = models.DateTimeField(auto_now_add=True)
    
    class Meta:
        db_table = 'parking_lot'
    
    def __str__(self):
        return f"{self.lot_name} ({self.total_spots} spots)"
    
    def available_spots(self):
        """Count available parking spots"""
        return self.total_spots - ParkedVehicle.objects.filter(
            parking_lot=self,
            checkout_time__isnull=True
        ).count()


class ParkingSpot(models.Model):
    """Individual parking spots"""
    spot_id = models.AutoField(primary_key=True)
    parking_lot = models.ForeignKey(ParkingLot, on_delete=models.CASCADE)
    spot_number = models.IntegerField()
    status = models.CharField(max_length=20, default='vacant')
    latitude = models.FloatField(null=True)
    longitude = models.FloatField(null=True)
    
    class Meta:
        db_table = 'parking_spot'
        unique_together = ['parking_lot', 'spot_number']
    
    def __str__(self):
        return f"Spot {self.spot_number} - {self.status}"


class Vehicle(models.Model):
    """Vehicle information"""
    vehicle_id = models.AutoField(primary_key=True)
    plate_number = models.CharField(max_length=20, unique=True)
    vehicle_type = models.CharField(max_length=50)
    entry_time = models.DateTimeField(auto_now_add=True)
    exit_time = models.DateTimeField(null=True, blank=True)
    
    class Meta:
        db_table = 'vehicle'
    
    def __str__(self):
        return self.plate_number


class ParkingRecord(models.Model):
    """Parking transaction records"""
    record_id = models.AutoField(primary_key=True)
    vehicle = models.ForeignKey(Vehicle, on_delete=models.CASCADE)
    parking_lot = models.ForeignKey(ParkingLot, on_delete=models.CASCADE)
    parking_spot = models.ForeignKey(ParkingSpot, on_delete=models.SET_NULL, 
                                     null=True)
    entry_time = models.DateTimeField(auto_now_add=True)
    exit_time = models.DateTimeField(null=True, blank=True)
    duration_minutes = models.IntegerField(null=True)
    status = models.CharField(max_length=20, default='parked')
    
    class Meta:
        db_table = 'parking_record'


class Upload_File(models.Model):
    """Video file uploads"""
    Video_Id = models.AutoField(primary_key=True)
    Video = models.FileField(upload_to='videos/')
    uploaded_at = models.DateTimeField(auto_now_add=True)
    processed = models.BooleanField(default=False)
    
    class Meta:
        db_table = 'upload_video'

Solution: Use smaller model (yolov8n.pt) or reduce video resolution

Solution: Ensure clear camera view, good lighting conditions

Solution: Delete db.sqlite3 and run migrations again

D.1 PRODUCTION DEPLOYMENT

Option 1: Deploy on Heroku

heroku login

3. Set environment variables:

heroku config:set DEBUG=False

heroku config:set ENABLE_YOLOV8=True

4. Deploy code:

git push heroku main

5. Run migrations:

heroku run python manage.py migrate

6. Create superuser:

heroku run python manage.py createsuperuser


Option 2: Deploy on AWS EC2

1. Launch EC2 instance (Ubuntu 20.04 LTS)

2. Connect and install dependencies:

sudo apt update
sudo apt install python3-pip python3-venv
sudo apt install git

3. Clone repository:

git clone https://github.com/jayaprakashroya/SmartSlot.git
cd SmartSlot

4. Setup virtual environment:

python3 -m venv venv
source venv/bin/activate
pip install -r Requirements.txt

5. Configure nginx:

sudo apt install nginx
# Configure reverse proxy to Django application

6. Run with Gunicorn:

pip install gunicorn
gunicorn ParkingProject.wsgi:application --bind 0.0.0.0:8000

FROM python:3.10-slim

WORKDIR /app

COPY Requirements.txt .
RUN pip install --no-cache-dir -r Requirements.txt

COPY . .

RUN python manage.py collectstatic --noinput

CMD ["gunicorn", "ParkingProject.wsgi:application", "--bind", "0.0.0.0:8000"]

docker-compose.yml:

version: '3.8'

tail -f logs/django.log

pip install --upgrade -r Requirements.txt

python manage.py clearsessions
